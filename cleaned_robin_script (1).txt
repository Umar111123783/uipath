def read_text_file(path: Path) -> str:
    """Read a text file, allowing a few common encodings."""
    for encoding in ("utf-8", "utf-8-sig", "cp1252"):
        try:
            return path.read_text(encoding=encoding)
        except UnicodeDecodeError:
            continue
    raise ValueError(f"Could not decode text file: {path.name}")


@st.cache_data(show_spinner=False)
def load_mapping_workbook(path_string: str, modified_time: float) -> str:
    """Convert every worksheet in the mapping workbook into readable CSV text."""
    del modified_time  # Used only to invalidate Streamlit's cache when the file changes.
    workbook = pd.read_excel(path_string, sheet_name=None, engine="openpyxl", dtype=str)
    sections = []
    for sheet_name, frame in workbook.items():
        frame = frame.fillna("")
        sections.append(f"### Worksheet: {sheet_name}\n{frame.to_csv(index=False)}")
    return "\n\n".join(sections)


@st.cache_data(show_spinner=False)
def load_local_knowledge() -> str:
    """Load all project-level knowledge sources used in the system message."""
    required_files = [MAPPING_FILE, *TEXT_KNOWLEDGE_FILES]
    missing = [path.name for path in required_files if not path.is_file()]
    if missing:
        raise FileNotFoundError(
            "Missing required project file(s): " + ", ".join(missing)
        )

    mapping_content = load_mapping_workbook(
        str(MAPPING_FILE), MAPPING_FILE.stat().st_mtime
    )

    text_sections = []
    for path in TEXT_KNOWLEDGE_FILES:
        text_sections.append(
            f"## File: {path.name}\n\n{read_text_file(path)}"
        )

    return (
        "# AA360 to Power Automate mapping workbook\n\n"
        f"{mapping_content}\n\n"
        "# Reference Robin and Power Automate Desktop examples\n\n"
        + "\n\n".join(text_sections)
    )


def extract_json_documents(uploaded_zip) -> list[tuple[str, object]]:
    """Safely read and parse all JSON files found in the uploaded ZIP."""
    zip_bytes = uploaded_zip.getvalue()
    if len(zip_bytes) > MAX_ZIP_SIZE_BYTES:
        raise ValueError("The uploaded ZIP exceeds the 50 MB size limit.")

    documents = []
    with zipfile.ZipFile(io.BytesIO(zip_bytes)) as archive:
        infos = [info for info in archive.infolist() if not info.is_dir()]
        total_size = sum(info.file_size for info in infos)
        if total_size > MAX_UNCOMPRESSED_BYTES:
            raise ValueError(
                "The ZIP expands beyond the 100 MB safety limit."
            )

        json_infos = []
        for info in infos:
            normalized = PurePosixPath(info.filename.replace("\\", "/"))
            if normalized.is_absolute() or ".." in normalized.parts:
                raise ValueError(f"Unsafe path found in ZIP: {info.filename}")
            if normalized.suffix.lower() == ".json":
                json_infos.append(info)

        if not json_infos:
            raise ValueError("No .json file was found in the uploaded ZIP.")

        for info in json_infos:
            raw = archive.read(info)
            try:
                text = raw.decode("utf-8-sig")
            except UnicodeDecodeError as exc:
                raise ValueError(
                    f"JSON file {info.filename} is not UTF-8 encoded."
                ) from exc
            try:
                parsed = json.loads(text)
            except json.JSONDecodeError as exc:
                raise ValueError(
                    f"Invalid JSON in {info.filename}: line {exc.lineno}, "
                    f"column {exc.colno}: {exc.msg}"
                ) from exc
            documents.append((info.filename, parsed))

    return documents


def _safe_filename(name: str) -> str:
    cleaned = "".join(ch if ch.isalnum() or ch in "._-" else "_" for ch in name)
    return cleaned.strip("._") or "automation_export"


def generate_analysis_workbook(uploaded_zip) -> dict:
    """Save the uploaded ZIP locally and run the unmodified analyzer pipeline."""
    zip_bytes = uploaded_zip.getvalue()
    if len(zip_bytes) > MAX_ZIP_SIZE_BYTES:
        raise ValueError("The uploaded ZIP exceeds the 50 MB size limit.")
    digest = hashlib.sha256(zip_bytes).hexdigest()[:12]
    local_zip = GENERATED_DIR / f"{_safe_filename(Path(uploaded_zip.name).stem)}_{digest}.zip"
    local_zip.write_bytes(zip_bytes)
    result = run_pipeline(str(local_zip))
    workbook_path = Path(result["xlsx_path"])
    if not workbook_path.is_file():
        raise RuntimeError("The analyzer did not create an Excel workbook.")
    return {
        "upload_hash": digest,
        "workbook_path": str(workbook_path),
        "workbook_name": workbook_path.name,
        "workbook_bytes": workbook_path.read_bytes(),
        "analyzer_result": result,
    }


def workbook_to_system_prompt(workbook_path: str) -> str:
    """Create a prompt containing data from every sheet in the generated workbook."""
    workbook = pd.read_excel(workbook_path, sheet_name=None, engine="openpyxl", dtype=str)
    sections = []
    for sheet_name, frame in workbook.items():
        sections.append(f"### Sheet: {sheet_name}\n{frame.fillna('').to_csv(index=False)}")
    print('Sections:::::::', sections)
    return (

        "Use only the workbook data below. Do not add assumptions or outside knowledge.\n"
        "Return plain text only, with a maximum of 500 characters.\n\n"
        + "\n\n".join(sections)
    )


def extract_plain_llm_text(response_data: dict) -> str:
    choices = response_data.get("choices")
    if not isinstance(choices, list) or not choices:
        raise ValueError("The LLM response did not contain any choices.")
    first = choices[0]
    content = first.get("message", {}).get("content")
    if isinstance(content, str):
        text = strip_markdown_fence(content).strip()
    elif isinstance(content, list):
        text = "\n".join(
            part.get("text", "") for part in content
            if isinstance(part, dict) and isinstance(part.get("text"), str)
        ).strip()
    else:
        text = str(first.get("text") or "").strip()
    if not text:
        raise ValueError("The LLM response did not contain generated text.")
    return text if len(text) <= 500 else text[:497].rstrip() + "..."


def call_functional_summary_llm(api_key: str, model_name: str, workbook_path: str) -> str:
    payload = {
        "model": model_name,
        "messages": [{"role": "system", "content": workbook_to_system_prompt(workbook_path)}],
    }
    headers = {"accept": "application/json", "API-Key": api_key, "Content-Type": "application/json"}
    try:
        response = requests.post(API_URL, headers=headers, json=payload, timeout=REQUEST_TIMEOUT_SECONDS)
    except requests.RequestException as exc:
        raise RuntimeError(f"Could not connect to the LLM endpoint: {exc}") from exc
    if not response.ok:
        raise RuntimeError(f"LLM request failed with HTTP {response.status_code}. {response.text.strip()[:1000]}")
    try:
        return extract_plain_llm_text(response.json())
    except ValueError as exc:
        raise RuntimeError("The LLM endpoint returned an invalid response.") from exc

def build_user_message(documents: list[tuple[str, object]]) -> str:
    parts = [
        "Convert the following Automation Anywhere export JSON into one complete "
        "Power Automate Desktop Robin script. Preserve the workflow logic, variables, "
        "conditions, loops, error handling, and action order as accurately as possible."
    ]
    for filename, parsed in documents:
        formatted = json.dumps(parsed, ensure_ascii=False, indent=2)
        parts.append(f"## Automation Anywhere JSON: {filename}\n{formatted}")
    return "\n\n".join(parts)


def strip_markdown_fence(text: str) -> str:
    """Remove an optional surrounding Markdown code fence."""
    cleaned = text.strip().replace("\r\n", "\n")
    lines = cleaned.splitlines()
    if lines and lines[0].strip().startswith("```"):
        lines = lines[1:]
    if lines and lines[-1].strip() == "```":
        lines = lines[:-1]
    return "\n".join(lines).strip()


def clean_robin_script(text: str) -> str:
    """Normalize the Robin script while preserving directly pasteable code."""
    cleaned = strip_markdown_fence(text)
    if not cleaned:
        raise ValueError("The LLM returned an empty Robin script.")
    return cleaned + "\n"


def parse_llm_output(raw_text: str) -> dict:
    """Validate the required summary/robin_script JSON response."""
    candidate = strip_markdown_fence(raw_text)
    try:
        result = json.loads(candidate)
    except json.JSONDecodeError as exc:
        raise ValueError(
            "The LLM did not return the required JSON output format. "
            f"JSON error at line {exc.lineno}, column {exc.colno}: {exc.msg}"
        ) from exc

    if not isinstance(result, dict):
        raise ValueError("The LLM output must be a JSON object.")

    if set(result) != {"summary", "robin_script"}:
        raise ValueError(
            'The LLM output must contain exactly "summary" and "robin_script".'
        )

    summary = result.get("summary")
    robin_script = result.get("robin_script")
    if not isinstance(summary, str) or not summary.strip():
        raise ValueError('The LLM output contains an invalid or empty "summary".')
    if len(summary.strip()) > 300:
        raise ValueError('The LLM "summary" exceeds the 300-character limit.')
    if not isinstance(robin_script, str) or not robin_script.strip():
        raise ValueError('The LLM output contains an invalid or empty "robin_script".')

    return {
        "summary": summary.strip(),
        "robin_script": clean_robin_script(robin_script),
    }


def extract_llm_text(response_data: dict) -> dict:
    """Extract and validate an OpenAI-compatible chat completion response."""
    choices = response_data.get("choices")
    if not isinstance(choices, list) or not choices:
        raise ValueError("The LLM response did not contain any choices.")

    first_choice = choices[0]
    message = first_choice.get("message", {})
    content = message.get("content")

    if isinstance(content, str) and content.strip():
        return parse_llm_output(content)

    if isinstance(content, list):
        text_parts = []
        for part in content:
            if isinstance(part, dict) and isinstance(part.get("text"), str):
                text_parts.append(part["text"])
        combined = "\n".join(text_parts).strip()
        if combined:
            return parse_llm_output(combined)

    fallback = first_choice.get("text")
    if isinstance(fallback, str) and fallback.strip():
        return parse_llm_output(fallback)

    raise ValueError("The LLM response did not contain generated text.")

def call_llm(api_key: str, model_name: str, system_message: str, user_message: str) -> dict:
    payload = {
        "model": model_name,
        "messages": [
            {"role": "system", "content": system_message},
            {"role": "user", "content": user_message},
        ],
    }
    headers = {
        "accept": "application/json",
        "API-Key": api_key,
        "Content-Type": "application/json",
    }

    try:
        response = requests.post(
            API_URL,
            headers=headers,
            json=payload,
            timeout=REQUEST_TIMEOUT_SECONDS,
        )
    except requests.RequestException as exc:
        raise RuntimeError(f"Could not connect to the LLM endpoint: {exc}") from exc

    if not response.ok:
        # Limit displayed server output because it may contain internal details.
        detail = response.text.strip()[:1000]
        raise RuntimeError(
            f"LLM request failed with HTTP {response.status_code}. {detail}"
        )

    try:
        response_data = response.json()
    except ValueError as exc:
        raise RuntimeError("The LLM endpoint returned a non-JSON response.") from exc

    return extract_llm_text(response_data)
